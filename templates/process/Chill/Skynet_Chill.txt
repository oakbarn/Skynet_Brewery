step "Chill"
ask "Hook up the chiller and open its cooling water. Press Done when the water runs."
@if Chill_Pump
{{Chill_Pump}} = true
@end
@if Wort_Probe
wait {{Wort_Probe}} <= rcp_Ferment_Temp
@else
ask "Press Done when the wort is down to " + rcp_Ferment_Temp + rcp_Temp_Units + "."
@end
@if Chill_Pump
{{Chill_Pump}} = false
@end
ask "Close the chiller water. Press Done when it is off."
print "Wort chilled to " + rcp_Ferment_Temp + rcp_Temp_Units
