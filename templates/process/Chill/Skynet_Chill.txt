//Skynet_Chill
//About: Chill the wort to the pitching temperature. With Whirlpool hops in the Recipe it stops at each Hop Stand that has hops (High, Middle, Low), the cooling water goes off for that stand's time, then the chill finishes. Whirlpool hops marked Hop Rocket go in the Hop Infuser, with the wort running through it. Use this or Whirlpool and Chill.

step "Hook up the chiller"
ask "Hook up the chiller. Press Done when it is on."
@if Chill_Pump
{{Chill_Pump}} = true
@end
if rcp_Hop_Stand_High == true
  step "Chill to Hop Stand High"
@if Chill_Water
{{Chill_Water}} = true
@else
ask "Open the chiller cooling water. Press Done when it runs."
@end
@if Wort_Probe
  wait {{Wort_Probe}} <= rcp_Hop_Stand_High_Temp
@else
  ask "Press Done when the wort is down to " + rcp_Hop_Stand_High_Temp + rcp_Temp_Units + " for Hop Stand High."
@end
  step "Hop Stand High"
@if Chill_Water
{{Chill_Water}} = false
@else
ask "Turn the chiller cooling water off. Press Done when it is off."
@end
  if rcp_Hop_Infuser == true
@if Hop_Infuser_Valve
    {{Hop_Infuser_Valve}} = true
@end
    ask "Load the Hop Stand High hops into the Hop Infuser and run the wort through it (" + rcp_Hop_Stand_High_Temp + rcp_Temp_Units + "). Press Done when it flows through the hops."
  else
    ask "Add the Hop Stand High hops now, at " + rcp_Hop_Stand_High_Temp + rcp_Temp_Units + ". Press Done when they are in."
  endif
  print "Hop Stand High for " + rcp_Hop_Stand_High_Minutes + " minutes, cooling water off"
  sleep rcp_Hop_Stand_High_Minutes * 60000
endif
if rcp_Hop_Stand_Middle == true
  step "Chill to Hop Stand Middle"
@if Chill_Water
{{Chill_Water}} = true
@else
ask "Open the chiller cooling water. Press Done when it runs."
@end
@if Wort_Probe
  wait {{Wort_Probe}} <= rcp_Hop_Stand_Middle_Temp
@else
  ask "Press Done when the wort is down to " + rcp_Hop_Stand_Middle_Temp + rcp_Temp_Units + " for Hop Stand Middle."
@end
  step "Hop Stand Middle"
@if Chill_Water
{{Chill_Water}} = false
@else
ask "Turn the chiller cooling water off. Press Done when it is off."
@end
  if rcp_Hop_Infuser == true
@if Hop_Infuser_Valve
    {{Hop_Infuser_Valve}} = true
@end
    ask "Load the Hop Stand Middle hops into the Hop Infuser and run the wort through it (" + rcp_Hop_Stand_Middle_Temp + rcp_Temp_Units + "). Press Done when it flows through the hops."
  else
    ask "Add the Hop Stand Middle hops now, at " + rcp_Hop_Stand_Middle_Temp + rcp_Temp_Units + ". Press Done when they are in."
  endif
  print "Hop Stand Middle for " + rcp_Hop_Stand_Middle_Minutes + " minutes, cooling water off"
  sleep rcp_Hop_Stand_Middle_Minutes * 60000
endif
if rcp_Hop_Stand_Low == true
  step "Chill to Hop Stand Low"
@if Chill_Water
{{Chill_Water}} = true
@else
ask "Open the chiller cooling water. Press Done when it runs."
@end
@if Wort_Probe
  wait {{Wort_Probe}} <= rcp_Hop_Stand_Low_Temp
@else
  ask "Press Done when the wort is down to " + rcp_Hop_Stand_Low_Temp + rcp_Temp_Units + " for Hop Stand Low."
@end
  step "Hop Stand Low"
@if Chill_Water
{{Chill_Water}} = false
@else
ask "Turn the chiller cooling water off. Press Done when it is off."
@end
  if rcp_Hop_Infuser == true
@if Hop_Infuser_Valve
    {{Hop_Infuser_Valve}} = true
@end
    ask "Load the Hop Stand Low hops into the Hop Infuser and run the wort through it (" + rcp_Hop_Stand_Low_Temp + rcp_Temp_Units + "). Press Done when it flows through the hops."
  else
    ask "Add the Hop Stand Low hops now, at " + rcp_Hop_Stand_Low_Temp + rcp_Temp_Units + ". Press Done when they are in."
  endif
  print "Hop Stand Low for " + rcp_Hop_Stand_Low_Minutes + " minutes, cooling water off"
  sleep rcp_Hop_Stand_Low_Minutes * 60000
endif
step "Chill"
@if Chill_Water
{{Chill_Water}} = true
@else
ask "Open the chiller cooling water. Press Done when it runs."
@end
@if Wort_Probe
wait {{Wort_Probe}} <= rcp_Ferment_Temp
@else
ask "Press Done when the wort is down to " + rcp_Ferment_Temp + rcp_Temp_Units + "."
@end
@if Chill_Pump
{{Chill_Pump}} = false
@end
@if Chill_Water
{{Chill_Water}} = false
@else
ask "Turn the chiller cooling water off. Press Done when it is off."
@end
if rcp_Hop_Infuser == true
@if Hop_Infuser_Valve
  {{Hop_Infuser_Valve}} = false
@else
  ask "Take the Hop Infuser out of line (or close its valves). Press Done when it is."
@end
endif
print "Wort chilled to " + rcp_Ferment_Temp + rcp_Temp_Units
