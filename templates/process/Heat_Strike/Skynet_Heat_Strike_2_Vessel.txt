step "Fill the kettle"
ask "Fill the boil kettle with " + rcp_Strike_Volume + " " + rcp_Volume_Units + " of strike water. Press Done when it is in."
step "Heat the strike water"
@if HLT_Heater
{{HLT_Heater}}.target = rcp_Strike_Temp
{{HLT_Heater}} = on
@else
ask "Turn the kettle heat on and set it to " + rcp_Strike_Temp + rcp_Temp_Units + ". Press Done when it is on."
@end
@if HLT_Probe
wait {{HLT_Probe}} >= rcp_Strike_Temp
@else
ask "Press Done when the strike water reaches " + rcp_Strike_Temp + rcp_Temp_Units + "."
@end
@if HLT_Heater
{{HLT_Heater}} = off
@end
print "Strike water at " + rcp_Strike_Temp + rcp_Temp_Units
