step "Fill the vessel"
ask "Put the grain basket in, and fill the vessel with " + skn_rcp_Strike_Volume + " " + skn_rcp_Volume_Units + " of water (all of it for a full volume mash). Press Done when it is in."
step "Heat the strike water"
@if HLT_Heater
{{HLT_Heater}}.setpoint = skn_rcp_Strike_Temp
{{HLT_Heater}} = on
@else
ask "Turn the heat on and set it to " + skn_rcp_Strike_Temp + skn_rcp_Temp_Units + ". Press Done when it is on."
@end
@if HLT_Probe
wait {{HLT_Probe}} >= skn_rcp_Strike_Temp
@else
ask "Press Done when the water reaches " + skn_rcp_Strike_Temp + skn_rcp_Temp_Units + "."
@end
hop_alert "Pre Boil"
print "Strike water at " + skn_rcp_Strike_Temp + skn_rcp_Temp_Units
