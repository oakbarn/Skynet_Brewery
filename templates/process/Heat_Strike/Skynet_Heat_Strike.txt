step "Fill the HLT"
ask "Fill the Hot Liquor Tank with at least " + skn_rcp_Strike_Volume + " " + skn_rcp_Volume_Units + " of strike water (plus sparge water if it heats in the same tank). Press Done when it is full."
step "Heat the strike water"
@if HLT_Heater
{{HLT_Heater}}.setpoint = skn_rcp_Strike_Temp
{{HLT_Heater}} = on
@else
ask "Turn the Hot Liquor Tank heat on and set it to " + skn_rcp_Strike_Temp + skn_rcp_Temp_Units + ". Press Done when it is on."
@end
@if HLT_Probe
wait {{HLT_Probe}} >= skn_rcp_Strike_Temp
@else
ask "Press Done when the strike water reaches " + skn_rcp_Strike_Temp + skn_rcp_Temp_Units + "."
@end
hop_alert "Pre Boil"
print "Strike water at " + skn_rcp_Strike_Temp + skn_rcp_Temp_Units
