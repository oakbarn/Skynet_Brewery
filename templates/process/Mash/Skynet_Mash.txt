@if Mash_Pump
{{Mash_Pump}} = true
@end
@each_mash_step
if skn_rcp_Mash_Steps >= {{n}}
  step "Mash step {{n}}"
  print "Mash step {{n}}: " + skn_rcp_Mash_Step_{{n}}_Name + " at " + skn_rcp_Mash_Step_{{n}}_Temp + skn_rcp_Temp_Units + " for " + skn_rcp_Mash_Step_{{n}}_Time
@if Mash_Heater
  {{Mash_Heater}}.setpoint = skn_rcp_Mash_Step_{{n}}_Temp
  {{Mash_Heater}} = on
@end
@if Mash_Probe
  wait {{Mash_Probe}} >= skn_rcp_Mash_Step_{{n}}_Temp - 1
@else
  ask "Bring the mash to " + skn_rcp_Mash_Step_{{n}}_Temp + skn_rcp_Temp_Units + " (" + skn_rcp_Mash_Step_{{n}}_Name + "). Press Done when it is there."
@end
@if Mash_Timer
  {{Mash_Timer}}.countdown = true
  {{Mash_Timer}} = skn_rcp_Mash_Step_{{n}}_Time
  start {{Mash_Timer}}
  if {{n}} == 1
    hop_alert "Mash"
  endif
  wait {{Mash_Timer}} <= 00:00:00
  stop {{Mash_Timer}}
@else
  if {{n}} == 1
    hop_alert "Mash"
  endif
  sleep skn_rcp_Mash_Step_{{n}}_Time * 1000
@end
endif
@end
step "Mash out"
if skn_rcp_Mash_Out_Temp > 0
@if Mash_Heater
  {{Mash_Heater}}.setpoint = skn_rcp_Mash_Out_Temp
@end
@if Mash_Probe
  wait {{Mash_Probe}} >= skn_rcp_Mash_Out_Temp - 1
@else
  ask "Bring the mash to " + skn_rcp_Mash_Out_Temp + skn_rcp_Temp_Units + " for the mash out. Press Done when it is there."
@end
endif
@if Mash_Heater
{{Mash_Heater}} = off
@end
@if Mash_Pump
{{Mash_Pump}} = false
@end
step "First wort"
hop_alert "1st Wort"
ask "Put any first wort hops in the boil kettle now. Then set the valves to run the first wort from the mash tun to the kettle. Press Done when it is running."
step "Sparge"
ask "Sparge with water at " + skn_rcp_Sparge_Temp + skn_rcp_Temp_Units + " from the Hot Liquor Tank, slowly (at least 10 minutes), and collect " + skn_rcp_Boil_Volume + " " + skn_rcp_Volume_Units + " of wort in the boil kettle. Turn the kettle heat on once the wort covers the heating element. Press Done when the kettle has it all."
@if HLT_Heater
{{HLT_Heater}} = off
@end
