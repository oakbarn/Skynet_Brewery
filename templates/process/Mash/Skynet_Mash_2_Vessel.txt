@if Mash_Pump
{{Mash_Pump}} = true
@end
step "Sparge water"
ask "While the mash rests, fill the empty kettle with the sparge water. Press Done when it is in."
@if HLT_Heater
{{HLT_Heater}}.setpoint = skn_rcp_Sparge_Temp
{{HLT_Heater}} = on
@end
@each_mash_step
if skn_rcp_Mash_Steps >= {{n}}
  step "Mash step {{n}}"
  print "Mash step {{n}}: " + skn_rcp_Mash_Step_{{n}}_Name + " at " + skn_rcp_Mash_Step_{{n}}_Temp + skn_rcp_Temp_Units + " for " + skn_rcp_Mash_Step_{{n}}_Time
@if Mash_Heater
  {{Mash_Heater}}.setpoint = skn_rcp_Mash_Step_{{n}}_Temp
  {{Mash_Heater}} = on
@end
@if Mash_Probe
  wait {{Mash_Probe}} >= skn_rcp_Mash_Step_{{n}}_Temp - 1
@else
  ask "Bring the mash to " + skn_rcp_Mash_Step_{{n}}_Temp + skn_rcp_Temp_Units + " (" + skn_rcp_Mash_Step_{{n}}_Name + "). Press Done when it is there."
@end
@if Mash_Timer
  {{Mash_Timer}}.countdown = true
  {{Mash_Timer}} = skn_rcp_Mash_Step_{{n}}_Time
  start {{Mash_Timer}}
  if {{n}} == 1
    hop_alert "Mash"
  endif
  wait {{Mash_Timer}} <= 00:00:00
  stop {{Mash_Timer}}
@else
  if {{n}} == 1
    hop_alert "Mash"
  endif
  sleep skn_rcp_Mash_Step_{{n}}_Time * 1000
@end
endif
@end
step "Mash out"
if skn_rcp_Mash_Out_Temp > 0
@if Mash_Probe
  wait {{Mash_Probe}} >= skn_rcp_Mash_Out_Temp - 1
@else
  ask "Bring the mash to " + skn_rcp_Mash_Out_Temp + skn_rcp_Temp_Units + " for the mash out. Press Done when it is there."
@end
endif
@if Mash_Pump
{{Mash_Pump}} = false
@end
@if HLT_Heater
{{HLT_Heater}} = off
@end
step "Sparge"
hop_alert "1st Wort"
ask "Move the hot sparge water out of the kettle (into a cooler or bucket, or pump it over the grain as you go), then run the wort into the kettle and sparge with it. Collect " + skn_rcp_Boil_Volume + " " + skn_rcp_Volume_Units + ". Press Done when the kettle has it all."
