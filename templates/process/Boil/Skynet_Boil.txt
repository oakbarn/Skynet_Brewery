step "Pre-boil gravity"
ask "Take a pre-boil gravity reading of the wort in the kettle and write it down. Press Done when you have it."
step "Up to the boil"
@if Boil_Heater
{{Boil_Heater}}.target = rcp_Boil_Start_Temp + 5
{{Boil_Heater}} = on
@else
ask "Turn the boil kettle heat on full. Press Done when it is on."
@end
@if Boil_Probe
wait {{Boil_Probe}} >= rcp_Boil_Start_Temp
@end
step "Hot break"
ask "Watch for the hot break and keep it from boiling over. Press Done only when it is at a rolling boil: the boil timer starts then."
step "Boil"
@if Boil_Timer
{{Boil_Timer}}.countdown = true
{{Boil_Timer}} = rcp_Boil_Time
start {{Boil_Timer}}
wait {{Boil_Timer}} <= 00:17:00
step "Sanitize the chiller"
ask "About 17 minutes left: put the chiller in the boil (or run hot wort through it) to sanitize it, and add the yeast nutrient and Whirlfloc at 15 minutes. Press Done when it is in."
wait {{Boil_Timer}} <= 00:00:00
stop {{Boil_Timer}}
@else
sleep rcp_Boil_Time * 1000
@end
step "Flame out"
@if Boil_Heater
{{Boil_Heater}} = off
@else
ask "Turn the boil kettle heat off. Press Done when it is off."
@end
print "Boil done"
