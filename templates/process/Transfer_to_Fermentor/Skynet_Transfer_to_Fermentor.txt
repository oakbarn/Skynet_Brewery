@if Transfer_Pump
ask "Set the valves from the kettle to the fermenter. Press Done when they are set."

@else
ask "Move the wort into the fermenter (about " + rcp_Batch_Volume + " " + rcp_Volume_Units + "). Press Done when it is in."
@end
close_all_valves
step "Gravity reading"
