//Clicking the advance switch (or Done on a Manual vessel that is waiting) adds 1 to gblV_Brew_Status.
new value vVBrew
[LoopAuto]
sleep 100
if sw_Advance_BrewStatus == true
	vVBrew = gblV_Brew_Status
	vVBrew += 1
	gblV_Brew_Status = vVBrew
	sw_Advance_BrewStatus = false
endif
if BrewZilla.confirmed == true
	BrewZilla.confirmed = false
	BrewZilla.waiting = false
	vVBrew = gblV_Brew_Status
	vVBrew += 1
	gblV_Brew_Status = vVBrew
endif
if DigiBoil.confirmed == true
	DigiBoil.confirmed = false
	DigiBoil.waiting = false
	vVBrew = gblV_Brew_Status
	vVBrew += 1
	gblV_Brew_Status = vVBrew
endif
goto "LoopAuto"
