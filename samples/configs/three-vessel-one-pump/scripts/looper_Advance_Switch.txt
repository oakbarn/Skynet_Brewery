//Clicking the advance switch (or Done on a Manual vessel that is waiting) adds 1 to gblV_Brew_Status.
new value vVBrew
[LoopAuto]
sleep 100
if "sw_Advance_BrewStatus" state == true
	vVBrew = "gblV_Brew_Status" value
	vVBrew += 1
	"gblV_Brew_Status" value = vVBrew
	"sw_Advance_BrewStatus" state = false
endif
goto "LoopAuto"
