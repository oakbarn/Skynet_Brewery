// All Fire Hose and Manifold Supply Valve Water on the Fly.
// OakBarn Brewery's script (07/27/2023, modified 06/27/2026), brought to Skynet for the 3 Vessel 1 Pump sample's
// Water Tab (Process Sample 1, 2026-11-20; Panel Update 9: the Water buttons act as one Group). His codes, labels and messages are kept; only what Skynet needs changed:
//   - a Process Widget on the Water Tab sets the code and starts this Process (no looper switch);
//   - "Manifold Set?" is an ask with Done (was the Advance switch and gblVEndlessLoop 799 / 800);
//   - Set Time is a pop-up that asks for the time into gblT_SetTimer_Water (was a Set Timer slot on his Water workspace);
//   - the run is one wait; the Water Override button (Rule 16.11) shuts the water off early;
//   - a hidden count-up timer, wgt_TMR_Water~elapsed_WA, keeps the true run time in gblT_Elapsed_Water (for seconds per gallon);
//   - his screen boxes (gblS_Brewery_Top_2 / _3, Bottom_1 / _3 / _4, the insp_ pictures) are one message box, gblS_Water_Text_1_Water.
// The code in gblV_Water_Switch_Status_Water:
//   1001 Fire Hose 1 Minute, 1002 2 Minutes, 1003 20 Seconds, 1004 Set Time, 1005 until the Float
//   2001 Manifold 1 Minute, 2002 2 Minutes, 2003 3 Minutes, 2004 Set Time, 2005 until the Float
//   999 or -9999: nothing to do
new string vSMsg
new string vSMsg1
new value vVSwitchNumber
new value vV_GBrew_Status
new string vSTimerDisplayName
new value vVPump
new string vSVesselFrom
new string vS_Label
new time vT_Set
vVSwitchNumber = gblV_Water_Switch_Status_Water
vV_GBrew_Status = gblV_Brew_Status
print "@16 Water Valve = "
print vVSwitchNumber
if vVSwitchNumber == -9999
	goto "End_FMS_Water"
endif
if vVSwitchNumber == 999
	goto "End_FMS_Water"
endif
BM_33_DO@Valve_E_Manifold~Supply_WA = false
BM_31_DO@Valve_E_Fire~Hose_WA = false
wgt_TMR_Water~delay_WA.visible = false
// the Water timer shows 00:00:00 from the press until the water flows (Panel Update 9, item 12d)
stop "wgt_TMR_Water_WA"
wgt_TMR_Water_WA = 00:00:00
gblS_Water_Text_1_Water = "Wait! Will Auto Advance"
//...........................................
//Set Water Valve Status
//FireHose
if vVSwitchNumber > 999
	if vVSwitchNumber < 2000
		print "@ 92 Firehose Timers"
		vSVesselFrom = "Fire Hose"
		vVPump = 200
		vSTimerDisplayName = "Firehose Timer"
		vS_Label = "Get Fire Hose Ready!"
		gblV_Water_Switch_Status_Water = -1
		vSMsg = "Fire Hose ON for "
		goto "Timed_FireHose"
	endif
endif
//...........................................
if vVSwitchNumber > 1999
	if vVSwitchNumber < 3000
		vSVesselFrom = "Manifold Supply Valve"
		vVPump = 100
		vSTimerDisplayName = "Manifold Supply On"
		vSMsg = "Manifold Supply Valve ON for "
		goto "Manifold_Timed"
	endif
endif
print "Not a Fire Hose or Manifold code"
goto "End_FMS_Water"
//..............................................
[Timed_FireHose]
print "@ 119  Timed_FireHose"
gblS_Water_Text_1_Water = "You may set the Manifold Manually how the transfer should be as needed. The Firehose can be used without setting any additional valves.  Simply set the manifold to the desired configuration by clicking on a valve to toggle between On and Off. You are the Safety and Control. The Brewery Pane and Valves will be restored to the way they were prior."
ask "Manifold Set?"
//FireHose 1 minutes
if vVSwitchNumber == 1001
	vT_Set = 00:01:00
	vSMsg1 = " One Minute. It will shut off when timer reaches 00:00:00!"
	vSMsg = vSMsg + vSMsg1
	goto "Delay_FireHose"
endif
//FireHose 2 minutes
if vVSwitchNumber == 1002
	vT_Set = 00:02:00
	vSMsg1 = " Two Minutes. "
	vSMsg = vSMsg + vSMsg1
	goto "Delay_FireHose"
endif
//FireHose 20 Seconds
if vVSwitchNumber == 1003
	vT_Set = 00:00:20
	vSMsg1 = " Twenty Seconds. "
	vSMsg = vSMsg + vSMsg1
	goto "Delay_FireHose"
endif
//Firehose Set Time
if vVSwitchNumber == 1004
	gblT_SetTimer_Water = 00:00:00
	gblS_Water_Text_1_Water = "Set the desired Run Time."
	// a pop-up asks for the time and puts it in the gGlobal gblT_SetTimer_Water (Rule 16.5); asked again if it is 00:00:00
	[Ask_Set_Time_FH]
	ask "Set the desired Run Time (hh:mm:ss)" gblT_SetTimer_Water
	if gblT_SetTimer_Water <= 00:00:00
		goto "Ask_Set_Time_FH"
	endif
	vT_Set = gblT_SetTimer_Water
	vSMsg1 = " "
	vSMsg = vSMsg + vSMsg1
	vSMsg1 = vT_Set
	vSMsg = vSMsg + vSMsg1
	vSMsg1 = ". "
	vSMsg = vSMsg + vSMsg1
	goto "Delay_FireHose"
endif
if vVSwitchNumber == 1005
	vT_Set = 00:00:00
	vSMsg1 = " The Firehose will be on until the FireHose Float is made.  A Timer will start and count up.  Click OverRide? to shut off the Firehose!"
	vSMsg = vSMsg + vSMsg1
	vSTimerDisplayName = "Fire Hose Float"
	goto "Delay_FireHose"
endif
goto "End_FMS_Water"
//................................................................
[Delay_FireHose]
print "@ 204 Delay_FireHose"
gblS_Water_Text_1_Water = "FireHose ON in 20 Seconds! " + vSMsg
stop "wgt_TMR_Water~delay_WA"
wgt_TMR_Water~delay_WA = 00:00:20
wgt_TMR_Water~delay_WA.countdown = true
wgt_TMR_Water~delay_WA.visible = true
start "wgt_TMR_Water~delay_WA"
wgt_ALM_Fire~Hose_WA = true
wait wgt_TMR_Water~delay_WA <= 00:00:10
gblS_Water_Text_1_Water = "FireHose ON in 10 Seconds!"
wait wgt_TMR_Water~delay_WA <= 00:00:05
gblS_Water_Text_1_Water = "FireHose ALMOST On!"
wait wgt_TMR_Water~delay_WA <= 00:00:00
stop "wgt_TMR_Water~delay_WA"
wgt_TMR_Water~delay_WA.visible = false
wgt_ALM_Fire~Hose_WA = false
gblS_Water_Text_1_Water = vSMsg
if vVSwitchNumber == 1005
	goto "Start_Timer_Floats"
endif
goto "Start_Timers"
//..............................................
[Manifold_Timed]
print "@ 256 Manifold Timed"
vS_Label = "Manifold Supply Valve will Open!"
gblS_Water_Text_1_Water = "You need to set the Manifold Manually how the transfer should be.  Simply set the manifold to the desired configuration by clicking on a valve to toggle between On and Off.  You are the Safety and Control. The Brewery Pane and Valves will be restored to the way they were prior."
ask "Manifold Set?"
//Manifold 1 minute
if vVSwitchNumber == 2001
	vT_Set = 00:01:00
	vSMsg1 = " One Minute. "
	vSMsg = vSMsg + vSMsg1
	goto "Delay_Manifold"
endif
//Manifold 2 minute
if vVSwitchNumber == 2002
	vT_Set = 00:02:00
	vSMsg1 = " Two Minutes. "
	vSMsg = vSMsg + vSMsg1
	goto "Delay_Manifold"
endif
//Manifold 3 minute
if vVSwitchNumber == 2003
	vT_Set = 00:03:00
	vSMsg1 = " Three Minutes. "
	vSMsg = vSMsg + vSMsg1
	goto "Delay_Manifold"
endif
if vVSwitchNumber == 2004
	gblT_SetTimer_Water = 00:00:00
	gblS_Water_Text_1_Water = "Set the desired Run Time."
	// a pop-up asks for the time and puts it in the gGlobal gblT_SetTimer_Water (Rule 16.5); asked again if it is 00:00:00
	[Ask_Set_Time_MSV]
	ask "Set the desired Run Time (hh:mm:ss)" gblT_SetTimer_Water
	if gblT_SetTimer_Water <= 00:00:00
		goto "Ask_Set_Time_MSV"
	endif
	vT_Set = gblT_SetTimer_Water
	vSMsg1 = " "
	vSMsg = vSMsg + vSMsg1
	vSMsg1 = vT_Set
	vSMsg = vSMsg + vSMsg1
	vSMsg1 = ". "
	vSMsg = vSMsg + vSMsg1
	goto "Delay_Manifold"
endif
if vVSwitchNumber == 2005
	vT_Set = 00:00:00
	vSMsg1 = " The Manifold Supply Valve will be open until the Main Brewery Float is made.  A Timer will start and count up.  Click OverRide? to shut the Manifold Supply Valve!"
	vSMsg = vSMsg + vSMsg1
	vSTimerDisplayName = "Main Brewery Float"
	goto "Delay_Manifold"
endif
goto "End_FMS_Water"
[Delay_Manifold]
print "@ 342 Delay Manifold"
gblS_Water_Text_1_Water = "Manifold Supply Valve ON in 20 Seconds! " + vSMsg
stop "wgt_TMR_Water~delay_WA"
wgt_TMR_Water~delay_WA = 00:00:20
wgt_TMR_Water~delay_WA.countdown = true
wgt_TMR_Water~delay_WA.visible = true
start "wgt_TMR_Water~delay_WA"
wgt_ALM_Water~flow_WA = true
wait wgt_TMR_Water~delay_WA <= 00:00:10
gblS_Water_Text_1_Water = "Manifold Supply Valve ON in 10 Seconds!"
wait wgt_TMR_Water~delay_WA <= 00:00:05
gblS_Water_Text_1_Water = "Manifold Supply Valve ALMOST On!"
wait wgt_TMR_Water~delay_WA <= 00:00:00
stop "wgt_TMR_Water~delay_WA"
wgt_TMR_Water~delay_WA.visible = false
wgt_ALM_Water~flow_WA = false
gblS_Water_Text_1_Water = vSMsg
if vVSwitchNumber == 2005
	goto "Start_Timer_Floats"
endif
goto "Start_Timers"
//................................................................
[Start_Timers]
print "@ 395 Start Timers"
stop "wgt_TMR_Water_WA"
wgt_TMR_Water_WA = vT_Set
wgt_TMR_Water_WA.countdown = true
start "wgt_TMR_Water_WA"
// a hidden count-up timer measures the true run time, from the valve opening to the wait being met or Override (for seconds per gallon)
stop "wgt_TMR_Water~elapsed_WA"
wgt_TMR_Water~elapsed_WA = 00:00:00
wgt_TMR_Water~elapsed_WA.countdown = false
start "wgt_TMR_Water~elapsed_WA"
if vVPump == 200
	// FireHose
	BM_31_DO@Valve_E_Fire~Hose_WA = true
endif
if vVPump == 100
	// Manifold Supply Valve
	BM_33_DO@Valve_E_Manifold~Supply_WA = true
endif
// the water flows: the button pressed shows Running, the other Water buttons stay greyed out (Panel Update 9, item 12a)
wgt_PSW_FireHose~1~Minute_WA.phase = "running"
// One wait: the timer reaching 00:00:00 ends it, or the Water Override button (Rule 16.11) shuts it off early
wait wgt_TMR_Water_WA <= 00:00:00
goto "Exit_Timer_End"
//................................................................
[Start_Timer_Floats]
print "@ 458 Start Timers for floats"
stop "wgt_TMR_Water_WA"
wgt_TMR_Water_WA = 00:00:00
wgt_TMR_Water_WA.countdown = false
start "wgt_TMR_Water_WA"
// a hidden count-up timer measures the true run time, from the valve opening to the wait being met or Override (for seconds per gallon)
stop "wgt_TMR_Water~elapsed_WA"
wgt_TMR_Water~elapsed_WA = 00:00:00
wgt_TMR_Water~elapsed_WA.countdown = false
start "wgt_TMR_Water~elapsed_WA"
if vVSwitchNumber == 1005
	BM_31_DO@Valve_E_Fire~Hose_WA = true
endif
if vVSwitchNumber == 2005
	BM_33_DO@Valve_E_Manifold~Supply_WA = true
endif
// the water flows: the button pressed shows Running, the other Water buttons stay greyed out (Panel Update 9, item 12a)
wgt_PSW_FireHose~1~Minute_WA.phase = "running"
// The float being made ends it, or Override shuts it off (his OverRide? here never worked: the quote in
// if "sw_Advance_Endless state" == true was in the wrong place)
wait BM_32_DI@Float_Water~float_WA == true
goto "Exit_Timer_End"
//....................................
[Exit_Timer_End]
print "@ 454 Exit_Timer_End"
stop "wgt_TMR_Water_WA"
// the true run time, whether the wait was met or Override ended it, is kept in gblT_Elapsed_Water
stop "wgt_TMR_Water~elapsed_WA"
gblT_Elapsed_Water = wgt_TMR_Water~elapsed_WA
if vVPump == 100
	BM_33_DO@Valve_E_Manifold~Supply_WA = false
endif
if vVPump == 200
	BM_31_DO@Valve_E_Fire~Hose_WA = false
endif
[End_FMS_Water]
print "@465 END"
wgt_ALM_Fire~Hose_WA = false
wgt_ALM_Water~flow_WA = false
wgt_TMR_Water~delay_WA.visible = false
gblS_Water_Text_1_Water = "Wait.  Will Auto Advance"
gblV_Brew_Status = vV_GBrew_Status
gblV_Water_Switch_Status_Water = -9999
BM_33_DO@Valve_E_Manifold~Supply_WA = false
BM_31_DO@Valve_E_Fire~Hose_WA = false
gblS_Water_Text_1_Water = "Ready. Pick a Fire Hose or Manifold button."
// every Water button shows Stopped and its Start works again (Panel Update 9, item 12a)
wgt_PSW_FireHose~1~Minute_WA.phase = "stopped"
