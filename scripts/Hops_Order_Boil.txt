//Orders all boil hops (group 919) by time, longest first.
//Hops with the same time get the same group number 1, 2, 3 ...
//Same loop as hops_HopOrder_Bitter, written for this program.
new value vVHopMax
new value vVSlot
new value vVN
new value vVT1
new value vVT2
new value vVT3
new value vVT4
new value vVT5
new value vVT6
new value vVT7
new value vVT8
vVT1 = -99
vVT2 = -99
vVT3 = -99
vVT4 = -99
vVT5 = -99
vVT6 = -99
vVT7 = -99
vVT8 = -99
	if "gblV_Hop1_Group_R1" value == 919
		vVT1 = "gblV_TimeHop1_Min_R1" value
	endif
	if "gblV_Hop2_Group_R1" value == 919
		vVT2 = "gblV_TimeHop2_Min_R1" value
	endif
	if "gblV_Hop3_Group_R1" value == 919
		vVT3 = "gblV_TimeHop3_Min_R1" value
	endif
	if "gblV_Hop4_Group_R1" value == 919
		vVT4 = "gblV_TimeHop4_Min_R1" value
	endif
	if "gblV_Hop5_Group_R1" value == 919
		vVT5 = "gblV_TimeHop5_Min_R1" value
	endif
	if "gblV_Hop6_Group_R1" value == 919
		vVT6 = "gblV_TimeHop6_Min_R1" value
	endif
	if "gblV_Hop7_Group_R1" value == 919
		vVT7 = "gblV_TimeHop7_Min_R1" value
	endif
	if "gblV_Hop8_Group_R1" value == 919
		vVT8 = "gblV_TimeHop8_Min_R1" value
	endif
vVSlot = 0
[Next_Slot]
vVHopMax = vVT1
	if vVT2 > vVHopMax
		vVHopMax = vVT2
	endif
	if vVT3 > vVHopMax
		vVHopMax = vVT3
	endif
	if vVT4 > vVHopMax
		vVHopMax = vVT4
	endif
	if vVT5 > vVHopMax
		vVHopMax = vVT5
	endif
	if vVT6 > vVHopMax
		vVHopMax = vVT6
	endif
	if vVT7 > vVHopMax
		vVHopMax = vVT7
	endif
	if vVT8 > vVHopMax
		vVHopMax = vVT8
	endif
	if vVHopMax == -99
		goto "Slots_Done"
	endif
vVSlot += 1
	if vVT1 == vVHopMax
		"gblV_Hop1_Group_R1" value = vVSlot
		vVT1 = -99
	endif
	if vVT2 == vVHopMax
		"gblV_Hop2_Group_R1" value = vVSlot
		vVT2 = -99
	endif
	if vVT3 == vVHopMax
		"gblV_Hop3_Group_R1" value = vVSlot
		vVT3 = -99
	endif
	if vVT4 == vVHopMax
		"gblV_Hop4_Group_R1" value = vVSlot
		vVT4 = -99
	endif
	if vVT5 == vVHopMax
		"gblV_Hop5_Group_R1" value = vVSlot
		vVT5 = -99
	endif
	if vVT6 == vVHopMax
		"gblV_Hop6_Group_R1" value = vVSlot
		vVT6 = -99
	endif
	if vVT7 == vVHopMax
		"gblV_Hop7_Group_R1" value = vVSlot
		vVT7 = -99
	endif
	if vVT8 == vVHopMax
		"gblV_Hop8_Group_R1" value = vVSlot
		vVT8 = -99
	endif
goto "Next_Slot"
[Slots_Done]
"gblVHop_Max" value = vVSlot
"shr_Hop_Slot" value = vVSlot
print "Boil hop groups: " + vVSlot
