//Whirlpool with cooling until the kettle reaches the hop stand
//temperature, cooling off, hop alarm, 20 minute stand, then chill.
//In Testing mode the stand is 20 seconds.
new value vV_Stand_Temp
new time vT_Stand
new string vSMsg
vV_Stand_Temp = 154
vT_Stand = 00:20:00
if sw_Status_Testing == true
	vT_Stand = 00:00:20
endif
BK_Burner = false
V_MLT_Out = false
V_BK_In = true
MB_36_do_RedPump_B1 = true
Cooling_Water = true
vSMsg = "Cooling to " + vV_Stand_Temp
vSMsg = vSMsg + " °F for the hop stand"
gblS_Inter_Script = vSMsg
wait BK_Temp <= vV_Stand_Temp
Cooling_Water = false
start "alm_Hops"
gblS_Inter_Script = "Add the whirlpool hops, then click the Hop alarm"
wait alm_Hops == false
tm_Whirlpool.countdown = true
tm_Whirlpool = vT_Stand
start "tm_Whirlpool"
gblS_Inter_Script = "Hop stand"
wait tm_Whirlpool <= 00:00:00
Cooling_Water = true
gblS_Inter_Script = "Hop stand done - chilling"
print "Hop stand finished at " + BK_Temp
